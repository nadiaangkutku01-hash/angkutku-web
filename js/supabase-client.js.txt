// ----------------------------------------------------------------------------
// Login -- pengganti window.loginUser() di Auth_Roles.html. BEDA PENTING:
// sistem lama pakai Username, Supabase Auth pakai Email. Kalau karyawan
// gak punya email asli, bisa dibuatkan email "buatan" seperti
// nama@angkutku.internal (gak perlu email sungguhan yang aktif).
// ----------------------------------------------------------------------------
async function loginUser(email, password) {
  const { data, error } = await supabaseClient.auth.signInWithPassword({ email, password });
  if (error) return { success: false, error: error.message };

  // Setelah login berhasil, ambil data Role/Nama dari tabel "users"
  // (disambungkan lewat kolom auth_user_id yang sudah kita siapkan di
  // Fase 2 Langkah A).
  const { data: userRow, error: userError } = await supabaseClient
    .from('users')
    .select('*')
    .eq('auth_user_id', data.user.id)
    .single();

  if (userError || !userRow) {
    return { success: false, error: 'Akun ini belum tersambung ke data karyawan/user. Hubungi Owner.' };
  }

  window.CURRENT_USER_SESSION = {
    userId: userRow.user_id,
    username: userRow.username,
    name: userRow.nama_lengkap,
    role: userRow.role,
    karyawanId: userRow.karyawan_id,
    investorId: userRow.investor_id,
  };

  return { success: true, session: window.CURRENT_USER_SESSION };
}

// ----------------------------------------------------------------------------
// Cek sesi yang masih aktif (misal user refresh halaman) -- dipanggil
// begitu index.html dimuat, sebelum render apa pun.
// ----------------------------------------------------------------------------
async function restoreSession() {
  const { data: { session } } = await supabaseClient.auth.getSession();
  if (!session) return false;

  const { data: userRow } = await supabaseClient
    .from('users')
    .select('*')
    .eq('auth_user_id', session.user.id)
    .single();

  if (!userRow) return false;

  window.CURRENT_USER_SESSION = {
    userId: userRow.user_id,
    username: userRow.username,
    name: userRow.nama_lengkap,
    role: userRow.role,
    karyawanId: userRow.karyawan_id,
    investorId: userRow.investor_id,
  };
  return true;
}

async function logoutUser() {
  await supabaseClient.auth.signOut();
  window.CURRENT_USER_SESSION = null;
  window.location.href = 'login.html';
}
